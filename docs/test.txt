from sklearn.metrics import (accuracy_score, precision_score, recall_score,
                             fbeta_score, matthews_corrcoef, confusion_matrix,
                             roc_auc_score, average_precision_score)

def report_metrics(y_true, y_prob, threshold, label):
    y_true = np.asarray(y_true).astype(int)
    y_prob = np.asarray(y_prob)
    pred = (y_prob >= threshold).astype(int)
    tn, fp, fn, tp = confusion_matrix(y_true, pred, labels=[0, 1]).ravel()
    print(f"\n=== {label} | threshold = {threshold:.2f} ===")
    print(f"n = {len(y_true)}, crystalline rate = {y_true.mean():.2f}")
    print(f"ROC-AUC   = {roc_auc_score(y_true, y_prob):.3f}   PR-AUC = {average_precision_score(y_true, y_prob):.3f}")
    print(f"Accuracy  = {accuracy_score(y_true, pred):.3f}")
    print(f"Precision = {precision_score(y_true, pred, zero_division=0):.3f}   (of those flagged, % truly crystalline)")
    print(f"Recall    = {recall_score(y_true, pred, zero_division=0):.3f}   (of truly crystalline, % caught)")
    print(f"F2        = {fbeta_score(y_true, pred, beta=2, zero_division=0):.3f}   MCC = {matthews_corrcoef(y_true, pred):.3f}")
    print(f"TN={tn} FP={fp} FN={fn} TP={tp}")

def plot_prob_hist(y_true, y_prob, threshold, title, filename):
    y_true = np.asarray(y_true).astype(int)
    y_prob = np.asarray(y_prob)
    bins = np.linspace(0, 1, 21)
    plt.figure(figsize=(6, 4))
    plt.hist(y_prob[y_true == 0], bins=bins, alpha=0.6, label="Stable (truth)")
    plt.hist(y_prob[y_true == 1], bins=bins, alpha=0.6, label="Crystalline (truth)")
    plt.axvline(threshold, color="gray", lw=1, ls="--", label=f"Threshold {threshold:.2f}")
    plt.xlabel("Predicted probability of crystallizing")
    plt.ylabel("Number of formulations")
    plt.title(title)
    plt.legend()
    plt.tight_layout()
    plt.savefig(filename, dpi=150)
    plt.close()
    print(f"Saved {filename}")
